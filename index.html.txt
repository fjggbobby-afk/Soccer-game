// Safe storage wrapper (works even in restricted iPad Quick Look viewers)
let memoryStats = { wins: 0, losses: 0, goals: 0, tackles: 0, shotsTaken: 0, shotsMade: 0 };
function loadStats() {
  try {
    const saved = localStorage.getItem("superSoccerStats");
    if (saved) memoryStats = JSON.parse(saved);
  } catch (e) {}
  return memoryStats;
}
let stats = loadStats();

function saveStats() {
  try {
    localStorage.setItem("superSoccerStats", JSON.stringify(stats));
  } catch (e) {}
  updateStatsUI();
}

function resetStats() {
  stats = { wins: 0, losses: 0, goals: 0, tackles: 0, shotsTaken: 0, shotsMade: 0 };
  saveStats();
}

function updateStatsUI() {
  document.getElementById("st-wins").textContent = stats.wins;
  document.getElementById("st-losses").textContent = stats.losses;
  document.getElementById("st-goals").textContent = stats.goals;
  document.getElementById("st-tackles").textContent = stats.tackles;
  const pct = stats.shotsTaken > 0 ? Math.round((stats.shotsMade / stats.shotsTaken) * 100) : 0;
  document.getElementById("st-shotpct").textContent = pct + "%";
  document.getElementById("st-diff").textContent = 1 + stats.wins;
}
